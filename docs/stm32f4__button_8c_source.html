74void port_button_init(uint32_t button_id)
75{
76 // Retrieve the button struct using the private function and the button ID
77 stm32f4_button_hw_t *p_button = _stm32f4_button_get(button_id);
78
79 if (p_button == NULL) // if the button doesn't exist, do nothing
80 return;
81
82 stm32f4_system_gpio_config( // Configure GPIO as input with no pull-up or pull-down resistors
83 p_button->p_port,
84 p_button->pin,
85 STM32F4_GPIO_MODE_IN,
86 STM32F4_GPIO_PUPDR_NOPULL
87 );
88
89 stm32f4_system_gpio_config_exti( // Configure external interrupt on both rising and falling edges
90 p_button->p_port,
91 p_button->pin,
92 STM32F4_TRIGGER_RISING_EDGE |
93 STM32F4_TRIGGER_FALLING_EDGE |
94 STM32F4_TRIGGER_ENABLE_INTERR_REQ
95 );
96
97 stm32f4_system_gpio_exti_enable( // Enable the interrupt line with priority 1 and subpriority 0
98 p_button->pin,
99 1, // Priority level
100 0 // Subpriority level
101 );
102}
