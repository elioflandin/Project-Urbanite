151static void do_start_up_measure(fsm_t *p_this) {

167static void do_stop_urbanite(fsm_t *p_this) {
168 fsm_urbanite_t *p_fsm = (fsm_urbanite_t *)p_this;
169 fsm_button_reset_duration(p_fsm->p_fsm_button);
170 fsm_ultrasound_stop(p_fsm->p_fsm_ultrasound_rear);
171 fsm_display_set_status(p_fsm->p_fsm_display_rear, false);
172 if (p_fsm->is_paused) {
173 p_fsm->is_paused = false;
174 }
175 printf("[URBANITE][%ld] Urbanite system OFF\n", port_system_get_millis());

186static void do_pause_display(fsm_t *p_this) {
187 fsm_urbanite_t *p_fsm = (fsm_urbanite_t *)p_this;
188 fsm_button_reset_duration(p_fsm->p_fsm_button);
189 p_fsm->is_paused = !p_fsm->is_paused;
190 fsm_display_set_status(p_fsm->p_fsm_display_rear, !p_fsm->is_paused);
191 if (p_fsm->is_paused)
192 printf("[URBANITE][%ld] Urbanite system display PAUSE\n", port_system_get_millis());
193 else
194 printf("[URBANITE][%ld] Urbanite system display RESUME\n", port_system_get_millis());
195}

206static void do_display_distance(fsm_t *p_this) {
207 fsm_urbanite_t *p_fsm = (fsm_urbanite_t *)p_this;
208 uint32_t distance_cm = fsm_ultrasound_get_distance(p_fsm->p_fsm_ultrasound_rear);
209 if (p_fsm->is_paused) {
210 if (distance_cm < WARNING_MIN_CM / 2) {
211 fsm_display_set_distance(p_fsm->p_fsm_display_rear, distance_cm);
212 fsm_display_set_status(p_fsm->p_fsm_display_rear, true);
213 } else {
214 fsm_display_set_status(p_fsm->p_fsm_display_rear, false);
215 }
216 } else {
217 fsm_display_set_distance(p_fsm->p_fsm_display_rear, distance_cm);
218 }
219 printf("[URBANITE][%ld] Distance: %ld cm\n", port_system_get_millis(), distance_cm);
220}
