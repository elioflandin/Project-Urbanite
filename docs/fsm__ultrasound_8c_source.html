165static void do_set_distance(fsm_t *p_this)
166{
167 fsm_ultrasound_t *p_us = (fsm_ultrasound_t *)p_this;
168
169 uint32_t init = port_ultrasound_get_echo_init_tick(p_us->ultrasound_id); // Step 1: Retrieve the echo init tick, echo end tick, and echo overflows from the ultrasound sensor
170 uint32_t end = port_ultrasound_get_echo_end_tick(p_us->ultrasound_id);
171 uint32_t overflows = port_ultrasound_get_echo_overflows(p_us->ultrasound_id);
172
173 uint32_t total_time_us = (end + (overflows * (0xFFFF))) - init; // Step 2: Calculate the time that the sound has taken to go back and forth
174
175 // Step3 : Calculate the distance in cm taking into account the speed of sound.
176 // SPEED_OF_SOUND_MS = 343 m/s => 0.0343 cm/us (divided by 10000) => one-way distance = (time / 2) * 0.0343
177 // uint32_t distance_cm = (uint32_t)(((double)total_time_us / 2.0) * ((double)SPEED_OF_SOUND_MS / 10000.0));
178 // uint32_t distance_cm = (uint32_t)((((double)total_time_us / 2.0) * ((double)SPEED_OF_SOUND_MS / 10000.0)) + 0.5);
179 uint32_t distance_cm = (total_time_us * SPEED_OF_SOUND_MS + 10000) / 20000; // Calculation without double
180
181 p_us->distance_arr[p_us->distance_idx] = distance_cm; // Step 4: Store the distance in the array of distances in the position of the index.
182 p_us->distance_idx++;
183 if (p_us->distance_idx >= FSM_ULTRASOUND_NUM_MEASUREMENTS) // Step 5: If the array is full, sort the array by calling the function qsort()
184 {
185 qsort(p_us->distance_arr, FSM_ULTRASOUND_NUM_MEASUREMENTS, sizeof(uint32_t), _compare);
186 // Step 6: If the array is full, compute the median of the array and store it in the field distance_cm, making sure to consider if FSM_ULTRASOUND_NUM_MEASUREMENTS is even or odd.
187 if (FSM_ULTRASOUND_NUM_MEASUREMENTS % 2 == 1)
188 {
189 p_us->distance_cm = p_us->distance_arr[FSM_ULTRASOUND_NUM_MEASUREMENTS / 2];
190 }
191 else
192 {
193 int i = FSM_ULTRASOUND_NUM_MEASUREMENTS / 2;
194 p_us->distance_cm = (p_us->distance_arr[i - 1] + p_us->distance_arr[i]) / 2;
195 }
196 p_us->new_measurement = true; // Step 7: If the array is full, set the flag new_measurement to indicate that a new measurement is ready.
197 p_us->distance_idx = 0; // Step 8: Increase the distance index. If the index is higher or equal FSM_ULTRASOUND_NUM_MEASUREMENTS, reset the index
198 }
199
200 port_ultrasound_stop_echo_timer(p_us->ultrasound_id); // Step 9: Call function port_ultrasound_stop_echo_timer to stop the timer that controls the input capture of the echo signal
201 port_ultrasound_reset_echo_ticks(p_us->ultrasound_id); // Step 10: Call function port_ultrasound_reset_echo_ticks() to reset the time ticks of the echo signal
202}
