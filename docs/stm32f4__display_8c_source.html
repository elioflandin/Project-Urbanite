141void port_display_init(uint32_t display_id)
142{
143 // 1. Retrieve the hardware configuration struct
144 stm32f4_display_hw_t *config = _stm32f4_display_get(display_id);
145 if (config == NULL) {
146 return; // If invalid display ID
147 }
148
149 // 2. Configure the GPIOs as alternate function, no pull-up or pull-down
150 const uint8_t MODE_AF = 2; // GPIO_MODE_AF = 2
151 const uint8_t NOPULL = 0; // GPIO_NOPULL = 0
152 stm32f4_system_gpio_config(config->p_port_red, config->pin_red, MODE_AF, NOPULL);
153 stm32f4_system_gpio_config(config->p_port_green, config->pin_green, MODE_AF, NOPULL);
154 stm32f4_system_gpio_config(config->p_port_blue, config->pin_blue, MODE_AF, NOPULL);
155
156 // 3. Configure the alternate function (AF2 for TIM4 on PB6, PB8, PB9)
157 const uint8_t af_tim4 = 2;
158 stm32f4_system_gpio_config_alternate(config->p_port_red, config->pin_red, af_tim4);
159 stm32f4_system_gpio_config_alternate(config->p_port_green, config->pin_green, af_tim4);
160 stm32f4_system_gpio_config_alternate(config->p_port_blue, config->pin_blue, af_tim4);
161
162 // 4. Configure the PWM timer (TIM4)
163 _timer_pwm_config(display_id);
164
165 // 5. Set the initial LED color to off
166 port_display_set_rgb(display_id, COLOR_OFF);
167 // volatile uint32_t cr1_final = TIM4->CR1; // Debug
168 // printf("CR1 after init = 0x%08lX\r\n", cr1_final);
169}

180void port_display_set_rgb(uint32_t display_id, rgb_color_t color)
181{
182 // Check that the display ID is supported
183 if (display_id != PORT_REAR_PARKING_DISPLAY_ID)
184 return;
185
186 TIM4->CR1 &= ~TIM_CR1_CEN; // Disable the timer before updating values
187
188 // If all components are zero, turn off all channels
189 if (color.r == 0 && color.g == 0 && color.b == 0) {
190 TIM4->CCER &= ~(TIM_CCER_CC1E | TIM_CCER_CC3E | TIM_CCER_CC4E); // Disable all channels
191 } else {
192 // === RED LED (Channel 1) ===
193 if (color.r == 0) {
194 TIM4->CCER &= ~TIM_CCER_CC1E; // Disable red channel
195 } else {
196 // Scale r from 0–255 to 0–ARR
197 TIM4->CCR1 = (uint32_t)(color.r * TIM4->ARR) / PORT_DISPLAY_RGB_MAX_VALUE;
198 TIM4->CCER |= TIM_CCER_CC1E; // Enable red channel
199 }
200
201 // === GREEN LED (Channel 3) ===
202 if (color.g == 0) {
203 TIM4->CCER &= ~TIM_CCER_CC3E; // Disable green channel
204 } else {
205 TIM4->CCR3 = (uint32_t)(color.g * TIM4->ARR) / PORT_DISPLAY_RGB_MAX_VALUE;
206 TIM4->CCER |= TIM_CCER_CC3E; // Enable green channel
207 }
208
209 // === BLUE LED (Channel 4) ===
210 if (color.b == 0) {
211 TIM4->CCER &= ~TIM_CCER_CC4E; // Disable blue channel
212 } else {
213 TIM4->CCR4 = (uint32_t)(color.b * TIM4->ARR) / PORT_DISPLAY_RGB_MAX_VALUE;
214 TIM4->CCER |= TIM_CCER_CC4E; // Enable blue channel
215 }
216 }
217
218 // Force update of registers and restart the timer, only if an LED is on
219 if (color.r != 0 || color.g != 0 || color.b != 0) {
220 TIM4->EGR |= TIM_EGR_UG; // Update generation
221 TIM4->CR1 |= TIM_CR1_CEN; // Re-enable the timer
222 }
223}
