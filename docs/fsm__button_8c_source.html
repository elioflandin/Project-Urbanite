111static void do_set_duration(fsm_t *p_this)
112{
113 fsm_button_t *p_fsm = (fsm_button_t *)p_this;
114 uint32_t now = port_system_get_millis();
115 p_fsm->duration = now - p_fsm->tick_pressed;
116 p_fsm->next_timeout = now + p_fsm->debounce_time;
117}

153{
