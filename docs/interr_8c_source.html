93void TIM2_IRQHandler(void)
94{
95 /* Resume the SysTick counter after timer/echo interruption (Version 4) */
96 port_system_systick_resume();
97
98 // Check if the update interrupt flag (UIF) is set -> overflow occurred
99 if (TIM2->SR & TIM_SR_UIF)
100 {
101 uint32_t current_overflows = port_ultrasound_get_echo_overflows(PORT_REAR_PARKING_SENSOR_ID); // Retrieve and increment the number of overflows
102 port_ultrasound_set_echo_overflows(PORT_REAR_PARKING_SENSOR_ID, current_overflows + 1);
103
104 TIM2->SR &= ~TIM_SR_UIF; // Clear the update interrupt flag
105 }
106
107 // Check if the input capture interrupt flag for channel 2 (CC2IF) is set
108 if (TIM2->SR & TIM_SR_CC2IF)
109 {
110 uint32_t current_tick = TIM2->CCR2; // Read the captured value from CCR2 (automatically clears CC2IF)
111
112 uint32_t echo_init = port_ultrasound_get_echo_init_tick(PORT_REAR_PARKING_SENSOR_ID); // Read current init and end ticks
113 uint32_t echo_end = port_ultrasound_get_echo_end_tick(PORT_REAR_PARKING_SENSOR_ID);
114
115 if (echo_init == 0 && echo_end == 0)
116 {
117 // First capture (rising edge): store the initial tick
118 port_ultrasound_set_echo_init_tick(PORT_REAR_PARKING_SENSOR_ID, current_tick);
119 }
120 else
121 {
122 // Second capture (falling edge): store the final tick and mark echo as received
123 port_ultrasound_set_echo_end_tick(PORT_REAR_PARKING_SENSOR_ID, current_tick);
124 port_ultrasound_set_echo_received(PORT_REAR_PARKING_SENSOR_ID, true);
125 }
126 }
127}
